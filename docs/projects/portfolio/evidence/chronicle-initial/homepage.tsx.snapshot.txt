  ProfilePreview,
  LabPreview,
  ContactClosing,

} from "@/components/semantic/homepage";

import { ChronicleCollection } from "./collection";

export function ChronicleHomepage({
  content,
  projects,
  locale,
  projectHref,

      <section

          id="selected-work-heading"

        <details>
          <summary>{content.about.label}</summary>

        </details>
        <details>
          <summary>{content.capabilities.title}</summary>
          <CapabilityList content={content.capabilities} />
        </details>
        <details>
          <summary>{content.lab.title}</summary>
          <LabPreview content={content.lab} locale={locale} />
        </details>
        <details>
          <summary>{content.contact.label}</summary>
          <ContactClosing content={content.contact} locale={locale} />
        </details>
      </div>
    </div>
