"use client";

import { useEffect, useRef, useState, type ReactNode } from "react";
import { ChronicleAtmosphere } from "./atmosphere";

// Server-rendered semantic cards/evidence cross the boundary; content never selects a theme.
export function ChronicleSelection({
  items,
  showPreview = true,
  showCardSelectors = true,
}: {
  items: { slug: string; title: string; card: ReactNode; preview: ReactNode }[];
  showPreview?: boolean;
  showCardSelectors?: boolean;
}) {
  const [selected, setSelected] = useState(0);
  const track = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const element = track.current;
    if (!element || !window.ResizeObserver) return;
    let width = element.clientWidth;
    const observer = new ResizeObserver(() => {
      if (element.clientWidth === width) return;
      width = element.clientWidth;
      const card = element.children[selected] as HTMLElement | undefined;
      if (card)
        element.scrollTo({
          left: card.offsetLeft - element.offsetLeft,
          behavior: "instant",
        });
    });
    observer.observe(element);
    return () => observer.disconnect();
  }, [selected]);
  const choose = (index: number) => {
    if (index < 0 || index >= items.length) return;
    setSelected(index);
    const card = track.current?.children[index] as HTMLElement | undefined;
    if (card && track.current)
      track.current.scrollTo({
        left: card.offsetLeft - track.current.offsetLeft,
        behavior: matchMedia("(prefers-reduced-motion: reduce)").matches
          ? "instant"
          : "smooth",
      });
  };
  if (!items.length) return null;
  return (
    <div
      className={`chronicle-selection${showPreview ? "" : " chronicle-selection--cards"}`}
    >
      <noscript>
        <style>{`.chronicle-selection__controls,.chronicle-selection__select,.chronicle-selection__card-target{display:none!important}`}</style>
      </noscript>
      <div
        className="chronicle-selection__controls"
        aria-label="Project selection"
        role="group"
      >
        <button
          type="button"
          aria-label="Previous project"
          onClick={() => choose(selected - 1)}
          disabled={selected === 0}
        >
          <svg viewBox="0 0 24 24" width="20" height="20" aria-hidden="true">
            <path
              d="m14 5-7 7 7 7"
              fill="none"
              stroke="currentColor"
              strokeWidth="1.5"
            />
          </svg>
        </button>
        <span role="status" aria-live="polite">
          {items[selected].title}
        </span>
        <div
          className="chronicle-selection__dots"
          role="group"
          aria-label="Choose a project"
        >
          {items.map((item, index) => (
            <button
              type="button"
              key={item.slug}
              aria-label={`Select ${item.title}`}
              aria-pressed={selected === index}
              onClick={() => choose(index)}
            >
              <span />
            </button>
          ))}
        </div>
        <button
          type="button"
          aria-label="Next project"
          onClick={() => choose(selected + 1)}
          disabled={selected === items.length - 1}
        >
          <svg viewBox="0 0 24 24" width="20" height="20" aria-hidden="true">
            <path
              d="m9 5 7 7-7 7"
              fill="none"
              stroke="currentColor"
              strokeWidth="1.5"
            />
          </svg>
        </button>
      </div>
      <div
        className="chronicle-selection__track"
        ref={track}
        tabIndex={0}
        aria-label="Project cards"
        onKeyDown={(event) => {
          if (event.target !== event.currentTarget) return;
          if (event.key === "ArrowRight" || event.key === "ArrowLeft") {
            event.preventDefault();
            choose(selected + (event.key === "ArrowRight" ? 1 : -1));
          }
        }}
        onScrollEnd={(event) => {
          const element = event.currentTarget;
          if (element.scrollWidth <= element.clientWidth + 2) return;
          const nearest = Array.from(element.children)
            .map((card, index) => ({
              index,
              distance: Math.abs(
                (card as HTMLElement).offsetLeft -
                  element.offsetLeft -
                  element.scrollLeft,
              ),
            }))
            .sort((a, b) => a.distance - b.distance)[0];
          if (nearest) setSelected(nearest.index);
        }}
      >
        {items.map((item, index) => (
          <div
            className="chronicle-selection__item"
            key={item.slug}
            data-selected={selected === index}
          >
            {showCardSelectors && (
              <button
                className="chronicle-selection__select"
                type="button"
                aria-pressed={selected === index}
                aria-controls={
                  showPreview ? `chronicle-preview-${item.slug}` : undefined
                }
                aria-label={`Preview ${item.title}`}
                onClick={() => choose(index)}
              >
                <span aria-hidden="true">◇</span>
                <span className="visually-hidden">Preview {item.title}</span>
              </button>
            )}
            {item.card}
            {showPreview && !showCardSelectors && (
              <button
                type="button"
                className="chronicle-selection__card-target"
                aria-label={`Preview ${item.title}`}
                aria-pressed={selected === index}
                aria-controls={`chronicle-preview-${item.slug}`}
                onClick={() => choose(index)}
              />
            )}
          </div>
        ))}
      </div>
      {showPreview && (
        <section
          className="chronicle-preview"
          aria-label="Selected project preview"
        >
          <ChronicleAtmosphere />
          {items.map((item, index) => (
            <div
              id={`chronicle-preview-${item.slug}`}
              key={item.slug}
              hidden={selected !== index}
              className="chronicle-preview__panel"
              data-motion-id={
                selected === index ? "selected-project-preview" : undefined
              }
            >
              {item.preview}
            </div>
          ))}
        </section>
      )}
    </div>
  );
}
